# RLS policies  

\# Phase 0 RLS Policies (Master Spec v2.2)

  

\#\# Core Principles

1\. \*\*Self-access by default\*\* - Users can always see their own records

2\. \*\*Home-scoped access\*\* - Home members can see home-related data

3\. \*\*School-scoped access\*\* - School access is controlled by capabilities

4\. \*\*No implicit visibility\*\* - Enrollment doesn't imply data access

5\. \*\*Emergency override\*\* - Temporary access during active emergencies

  

\---

  

\#\# 1. Core Identity Tables

  

\#\#\# persons

\`\`\`sql

\-- Enable RLS

ALTER TABLE persons ENABLE ROW LEVEL SECURITY;

  

\-- Policy 1: Self-access

CREATE POLICY "Users can view themselves"

ON persons FOR SELECT

USING (id = auth.uid ());

  

\-- Policy 2: Home members can view each other

CREATE POLICY "Home members can view co-members"

ON persons FOR SELECT

USING (

  EXISTS (

    SELECT 1 FROM home\_memberships hm1

    JOIN home\_memberships hm2 ON hm1.home\_id = hm2.home\_id

    WHERE hm1.person\_id = persons.id

    AND hm2.person\_id = auth.uid ()

  )

);

  

\-- Policy 3: School staff with capability can view students

CREATE POLICY "School staff with capability can view enrolled persons"

ON persons FOR SELECT

USING (

  EXISTS (

    SELECT 1 FROM enrollments e

    JOIN school\_capability\_grants scg ON e.school\_id = scg.school\_id

    WHERE e.person\_id = persons.id

    AND scg.person\_id = auth.uid ()

    AND scg.capability\_key LIKE 'school.enrollments.view%'

  )

);

  

\-- Policy 4: Emergency access (temporary override)

CREATE POLICY "Emergency access to person"

ON persons FOR SELECT

USING (

  EXISTS (

    SELECT 1 FROM emergency\_events ee

    WHERE ee.subject\_person\_id = persons.id

    AND ee.status = 'active'

    AND ee.school\_id IN (

      SELECT school\_id FROM positions 

      WHERE person\_id = auth.uid () AND status = 'confirmed'

    )

  )

);

\`\`\`sql

ALTER TABLE person\_information ENABLE ROW LEVEL SECURITY;

  

\-- Policy 1: Self-access (full)

CREATE POLICY "Users can view their own information"

ON person\_information FOR SELECT

USING (person\_id = auth.uid ());

  

\-- Policy 2: Self-edit

CREATE POLICY "Users can update their own information"

ON person\_information FOR UPDATE

USING (person\_id = auth.uid ());

  

\-- Policy 3: Home primary can view home members' information

CREATE POLICY "Home primary can view member information"

ON person\_information FOR SELECT

USING (

  EXISTS (

    SELECT 1 FROM home\_memberships hm

    JOIN homes h ON hm.home\_id = h.id

    WHERE hm.person\_id = person\_information.person\_id

    AND h.primary\_person\_id = auth.uid ()

  )

);

  

\-- Policy 4: School access via explicit grants

CREATE POLICY "School staff with field-level grants can view"

ON person\_information FOR SELECT

USING (

  EXISTS (

    SELECT 1 FROM person\_information\_access pia

    LEFT JOIN enrollments e ON pia.enrollment\_id = e.id

    LEFT JOIN positions p ON pia.position\_id = p.id

    WHERE (e.person\_id = person\_information.person\_id OR p.person\_id = person\_information.person\_id)

    AND (

      (e.id IS NOT NULL AND e.school\_id IN (

        SELECT school\_id FROM positions WHERE person\_id = auth.uid ()

      ))

      OR

      (p.id IS NOT NULL AND p.school\_id IN (

        SELECT school\_id FROM positions WHERE person\_id = auth.uid ()

      ))

    )

    -- Note: Actual field-level filtering happens in application logic

    -- This just allows access to the row

  )

);

  

\-- Policy 5: Emergency access (full information during emergency)

CREATE POLICY "Emergency access to person information"

ON person\_information FOR SELECT

USING (

  EXISTS (

    SELECT 1 FROM emergency\_events ee

    WHERE ee.subject\_person\_id = person\_information.person\_id

    AND ee.status = 'active'

    AND ee.school\_id IN (

      SELECT school\_id FROM positions 

      WHERE person\_id = auth.uid () AND status = 'confirmed'

    )

  )

);

\#\# 10. Helper Functions for RLS

  

Create these helper functions to make RLS policies cleaner:

  

\`\`\`sql

\-- Check if user has capability at a school

CREATE OR REPLACE FUNCTION has\_capability(

  check\_person\_id UUID,

  check\_school\_id UUID,

  required\_capability\_pattern TEXT

)

RETURNS BOOLEAN AS $$

BEGIN

  RETURN EXISTS (

    SELECT 1 FROM school\_capability\_grants scg

    WHERE scg.person\_id = check\_person\_id

    AND scg.school\_id = check\_school\_id

    AND scg.capability\_key LIKE required\_capability\_pattern

    AND NOT EXISTS (

      SELECT 1 FROM school\_capability\_denies scd

      WHERE scd.person\_id = check\_person\_id

      AND scd.school\_id = check\_school\_id

      AND scd.capability\_key LIKE required\_capability\_pattern

    )

  );

END;

$$ LANGUAGE plpgsql SECURITY DEFINER;

  

\-- Check if user is home primary for a person

CREATE OR REPLACE FUNCTION is\_home\_primary\_for(

  check\_person\_id UUID,

  target\_person\_id UUID

)

RETURNS BOOLEAN AS $$

BEGIN

  RETURN EXISTS (

    SELECT 1 FROM home\_memberships hm1

    JOIN home\_memberships hm2 ON hm1.home\_id = hm2.home\_id

    JOIN homes h ON hm1.home\_id = h.id

    WHERE hm1.person\_id = target\_person\_id

    AND hm2.person\_id = check\_person\_id

    AND h.primary\_person\_id = check\_person\_id

  );

END;

$$ LANGUAGE plpgsql SECURITY DEFINER;

  

\-- Check if users share a section (for teacher-student relationship)

CREATE OR REPLACE FUNCTION share\_section(

  person1\_id UUID,

  person2\_id UUID

)

RETURNS BOOLEAN AS $$

BEGIN

  RETURN EXISTS (

    SELECT 1 FROM section\_student\_memberships ssm

    JOIN section\_teacher\_memberships stm ON ssm.section\_id = stm.section\_id

    WHERE ssm.person\_id = person2\_id

    AND stm.person\_id = person1\_id

    AND ssm.status = 'active'

    AND stm.ended\_at IS NULL

  );

END;

$$ LANGUAGE plpgsql SECURITY DEFINER;

\#\# Summary

  

These RLS policies enforce:

  

1\. \*\*Self-access\*\* - Users can always see themselves

2\. \*\*Home boundaries\*\* - Home members share access

3\. \*\*School capabilities\*\* - Wildcard-based permission system

4\. \*\*Emergency override\*\* - Temporary access during emergencies

5\. \*\*No implicit visibility\*\* - Explicit grants required for personal data

  

The policies work together with the capability system to create a secure, flexible access control layer that matches your master spec exactly.
