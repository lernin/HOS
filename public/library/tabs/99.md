def process\_grow\_task(task\_id, target\_id, depth, parent\_id):

    hyponyms = generate\_hyponyms(parent\['string'\])

    

    if not hyponyms:

        mark\_terminal(target\_id)

        mark\_task\_completed(task\_id)

        propagate\_rooted(target\_id)

        print(f"  → Terminal: {parent\['string'\]}")

        return

    for hyponym in hyponyms:

def phase2\_create\_ontonyms():

    """Create ontonyms that mirror the English eidonym tree."""

    

    conn = get\_db\_connection()

    cur = conn.cursor()

    

    \# Get all English atomonyms in breadth-first order

    cur.execute("""

        WITH RECURSIVE english\_tree AS (

            SELECT id, string, hypernym\_id, 1 as depth

            FROM atomonyms

            WHERE language = 'en-US' AND hypernym\_id IS NULL

            

            UNION ALL

            

            SELECT a.id, a.string, a.hypernym\_id, et.depth + 1

            FROM atomonyms a

            JOIN english\_tree et ON a.hypernym\_id = et.id

            WHERE a.language = 'en-US'

        )

        SELECT id, string, hypernym\_id FROM english\_tree

        ORDER BY depth, id

    """)

    

    english\_atomonyms = cur.fetchall()

    

    \# Map English atomonym IDs to ontonym IDs

    atomonym\_to\_ontonym = {}

    

    for atomonym\_id, string, hypernym\_id in english\_atomonyms:

        \# Find parent ontonym (if any)

        parent\_ontonym\_id = None

        if hypernym\_id and hypernym\_id in atomonym\_to\_ontonym:

            parent\_ontonym\_id = atomonym\_to\_ontonym\[hypernym\_id\]

        

        \# Create ontonym

        cur.execute("SELECT COALESCE(MAX(id), 0) + 1 FROM ontonyms")

        next\_id = cur.fetchone()\[0\]

        

        cur.execute("""

            INSERT INTO ontonyms (id, string, hypernym\_id, status, confidence, stage)

            VALUES (%s, %s, %s, 'reviewed', 1.0, 5)

            RETURNING id

        """, (next\_id, string, parent\_ontonym\_id))

        

        ontonym\_id = cur.fetchone()\[0\]

        atomonym\_to\_ontonym\[atomonym\_id\] = ontonym\_id

        

        \# Link English atomonym to ontonym

        cur.execute("""

            UPDATE atomonyms 

            SET ontonym\_id = %s 

            WHERE id = %s

        """, (ontonym\_id, atomonym\_id))

        

        print(f"Linked: {string} (A:{atomonym\_id}) → O:{ontonym\_id}")

    

    conn.commit()

    cur.close()

    conn.close()

    

    print(f"\\nCreated {len(atomonym\_to\_ontonym)} ontonyms")

def phase3\_attach\_korean():

    """Attach Korean atomonyms to existing ontonyms."""

    

    conn = get\_db\_connection()

    cur = conn.cursor()

    

    \# Get all ontonyms in breadth-first order

    cur.execute("""

        WITH RECURSIVE ontonym\_tree AS (

            SELECT id, string, hypernym\_id, 1 as depth

            FROM ontonyms

            WHERE hypernym\_id IS NULL

            

            UNION ALL

            

            SELECT o.id, o.string, o.hypernym\_id, ot.depth + 1

            FROM ontonyms o

            JOIN ontonym\_tree ot ON o.hypernym\_id = ot.id

        )

        SELECT id, string, hypernym\_id FROM ontonym\_tree

        ORDER BY depth, id

    """)

    

    ontonyms = cur.fetchall()

    

    for ontonym\_id, concept, hypernym\_id in ontonyms:

        \# Check if Korean already exists

        cur.execute("""

            SELECT id FROM atomonyms 

            WHERE ontonym\_id = %s AND language = 'ko'

        """, (ontonym\_id,))

        

        if cur.fetchone():

            print(f"Korean already exists for: {concept}")

            continue

        

        \# Translate concept to Korean

        korean\_word = ai\_translate(concept, 'ko')

        

        if korean\_word and korean\_word \!= '':

            \# Find Korean hypernym

            korean\_hypernym\_id = None

            if hypernym\_id:

                cur.execute("""

                    SELECT id FROM atomonyms 

                    WHERE ontonym\_id = %s AND language = 'ko'

                """, (hypernym\_id,))

                row = cur.fetchone()

                if row:

                    korean\_hypernym\_id = row\[0\]

            

            \# Create Korean atomonym

            cur.execute("SELECT COALESCE(MAX(id), 0) + 1 FROM atomonyms")

            next\_id = cur.fetchone()\[0\]

            

            cur.execute("""

                INSERT INTO atomonyms (id, string, language, ontonym\_id, hypernym\_id, status, confidence)

                VALUES (%s, %s, 'ko', %s, %s, 'initialized', 0.7)

            """, (next\_id, korean\_word, ontonym\_id, korean\_hypernym\_id))

            

            print(f"Added Korean: {concept} → {korean\_word}")

        else:

            \# No direct translation — create hole

            description = ai\_describe(concept, 'ko')

            

            cur.execute("SELECT COALESCE(MAX(id), 0) + 1 FROM holes")

            hole\_id = cur.fetchone()\[0\]

            

            cur.execute("""

                INSERT INTO holes (id, ontonym\_id, missing\_language, source\_language, proposed\_pleronym, status)

                VALUES (%s, %s, 'ko', 'en', %s, 'open')

            """, (hole\_id, ontonym\_id, description))

            

            print(f"Hole created for: {concept} (Korean missing)")

    

    conn.commit()

    cur.close()

    conn.close()

def phase3\_grow\_korean\_only():

    """Add Korean-only concepts that don't exist in English."""

    

    conn = get\_db\_connection()

    cur = conn.cursor()

    

    \# Identify Korean concepts with no ontonym link

    \# (In a real system, you'd have a way to identify these)

    \# For MVP, we'll use a list or AI detection

    

    \# Example: Korean concept "정" (jeong)

    korean\_only\_concepts = \[

        {'word': '정', 'hypernym\_korean': '애착', 'description': 'deep emotional bond from shared experience'}

    \]

    

    for concept in korean\_only\_concepts:

        \# Find Korean hypernym atomonym

        cur.execute("""

            SELECT id FROM atomonyms 

            WHERE string = %s AND language = 'ko'

        """, (concept\['hypernym\_korean'\],))

        

        hypernym\_row = cur.fetchone()

        

        \# Find or create ontonym for hypernym

        cur.execute("""

            SELECT ontonym\_id FROM atomonyms 

            WHERE string = %s AND language = 'ko'

        """, (concept\['hypernym\_korean'\],))

        

        hypernym\_ontonym = cur.fetchone()

        

        \# Create new ontonym

        cur.execute("SELECT COALESCE(MAX(id), 0) + 1 FROM ontonyms")

        new\_ontonym\_id = cur.fetchone()\[0\]

        

        cur.execute("""

            INSERT INTO ontonyms (id, string, hypernym\_id, status, confidence)

            VALUES (%s, %s, %s, 'provisional', 0.7)

        """, (new\_ontonym\_id, concept\['word'\], hypernym\_ontonym\[0\] if hypernym\_ontonym else None))

        

        \# Create Korean atomonym

        cur.execute("SELECT COALESCE(MAX(id), 0) + 1 FROM atomonyms")

        new\_atomonym\_id = cur.fetchone()\[0\]

        

        cur.execute("""

            INSERT INTO atomonyms (id, string, language, ontonym\_id, hypernym\_id, status, confidence)

            VALUES (%s, %s, 'ko', %s, %s, 'initialized', 0.9)

        """, (new\_atomonym\_id, concept\['word'\], new\_ontonym\_id, hypernym\_row\[0\] if hypernym\_row else None))

        

        \# Create hole for English

        cur.execute("SELECT COALESCE(MAX(id), 0) + 1 FROM holes")

        hole\_id = cur.fetchone()\[0\]

        

        cur.execute("""

            INSERT INTO holes (id, ontonym\_id, missing\_language, source\_language, proposed\_pleronym, status)

            VALUES (%s, %s, 'en', 'ko', %s, 'open')

        """, (hole\_id, new\_ontonym\_id, concept\['description'\]))

        

        print(f"Added Korean-only concept: {concept\['word'\]} → new ontonym O:{new\_ontonym\_id}")

    

    conn.commit()

    cur.close()

    conn.close()
