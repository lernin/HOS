Minimal Pipeline Example

import spacy  
from pywsd.lesk import simple\_lesk

nlp \= spacy.load("en\_core\_web\_sm")

text \= "I drank milk yesterday."

doc \= nlp(text)

for token in doc:  
    if token.is\_alpha:  
        sense \= simple\_lesk(text, token.text, pos=token.pos\_\[0\].lower())

        print({  
            "token": token.text,  
            "lemma": token.lemma\_,  
            "pos": token.pos\_,  
            "sense": str(sense)  
        })
